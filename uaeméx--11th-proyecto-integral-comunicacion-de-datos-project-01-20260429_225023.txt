// ── Gantt chart ──────────────────────────────────────────────────────────────
#let gantt-chart(phases, total-days) = {
  let bar-height = 14pt
  let label-width = 160pt
  let chart-width = 300pt
  let row-gap = 6pt

  // Header: day ticks
  let tick-every = 5
  let ticks = range(0, total-days + 1, step: tick-every)

  stack(
    spacing: 0pt,

    // Tick labels
    stack(dir: ltr, box(width: label-width, ""), ..ticks.map(t => box(
      width: chart-width * (tick-every / total-days),
      text(size: 7pt, fill: gris-texto, "día " + str(t)),
    ))),

    v(4pt),

    // Bars
    ..phases.map(phase => {
      let x = phase.start / total-days
      let w = phase.duration / total-days
      stack(
        dir: ltr,
        spacing: 0pt,
        // Label
        box(width: label-width, height: bar-height + row-gap, align(horizon, text(size: size-small, phase.name))),
        // Empty space before bar
        box(width: chart-width * x, height: bar-height + row-gap),
        // The bar itself
        box(
          width: chart-width * w,
          height: bar-height,
          fill: azul-medio.lighten(40%),
          stroke: azul-medio,
          radius: 3pt,
          inset: (x: 4pt),
          align(horizon, text(size: 7pt, fill: azul-oscuro, phase.owner)),
        ),
      )
    }),
  )
}

#let category-table(category, items) = {
  let subtotal = items.map(i => i.qty * i.unit_cost).sum()

  block(breakable: false)[
    #text(weight: "bold", size: 11pt, fill: azul-medio, category)
    #v(4pt)
    #table(
      columns: (auto, 1fr, auto, auto, auto, auto),
      fill: (col, row) => if row == 0 { azul-fondo } else if calc.odd(row) { gris-claro } else { white },
      stroke: (x, y) => if y == 0 { (bottom: 0.5pt + rgb("#b0bec5")) } else { 0pt },
      inset: 8pt,
      align: (col, row) => if col > 1 { right } else { left },

      // Header
      text(weight: "bold", size: 9pt, "ID"),
      text(weight: "bold", size: 9pt, "Descripción"),
      text(weight: "bold", size: 9pt, "Cant."),
      text(weight: "bold", size: 9pt, "Unidad"),
      text(weight: "bold", size: 9pt, "Unitario"),
      text(weight: "bold", size: 9pt, "Total"),

      // Rows
      ..items
        .map(item => (
          text(size: 9pt, item.id),
          text(size: 9pt, item.description),
          str(item.qty),
          item.unit,
          format-currency(item.unit_cost),
          text(weight: "bold", format-currency(item.qty * item.unit_cost)),
        ))
        .flatten(),

      // Footer Subtotal
      table.cell(colspan: 5, align(right)[#text(weight: "bold", "Subtotal " + category)]),
      text(weight: "bold", fill: azul-medio, format-currency(subtotal)),
    )
    #v(12pt)
  ]
}
